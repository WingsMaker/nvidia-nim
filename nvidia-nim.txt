const apiKey = PropertiesService.getScriptProperties().getProperty("nvapi_key");
function callNvidiaNim() {
  const url = 'https://integrate.api.nvidia.com/v1/chat/completions';
  const payload = {
    model: 'minimaxai/minimax-m3', 
    messages: [
      {
        role: 'user',
        content: "translate 日落西山你不陪,东山再起你是谁 to english with profound meaning"
      }
    ],
    temperature: 0.2,
    top_p: 0.7,
    max_tokens: 1024
  };

  const options = {
    method: 'post',
    contentType: 'application/json',
    headers: {
      'Authorization': 'Bearer ' + apiKey
    },
    payload: JSON.stringify(payload),
    muteHttpExceptions: true
  };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const responseCode = response.getResponseCode();
    const responseBody = response.getContentText();

    if (responseCode === 200) {
      const jsonResponse = JSON.parse(responseBody);
      const outputText = jsonResponse.choices[0].message.content;
      Logger.log('回應內容：\n' + outputText);
      return outputText;
    } else {
      Logger.log('錯誤代碼：' + responseCode);
      Logger.log('錯誤詳細資訊：' + responseBody);
    }
  } catch (e) {
    Logger.log('發生例外狀況：' + e.toString());
  }
}
